cmake_minimum_required(VERSION 3.20)
project(tokenscope_tracer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.runtime/llama.cpp" CACHE PATH "Pinned llama.cpp source")
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(tokenscope-tracer tracer.cpp)
target_include_directories(tokenscope-tracer PRIVATE "${LLAMA_SOURCE}/src" "${LLAMA_SOURCE}/vendor")
target_link_libraries(tokenscope-tracer PRIVATE llama llama-common ggml ggml-cpu)
target_compile_definitions(tokenscope-tracer PRIVATE TOKENSCOPE_LLAMA_SOURCE_DIR="${LLAMA_SOURCE}")
set_target_properties(tokenscope-tracer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
